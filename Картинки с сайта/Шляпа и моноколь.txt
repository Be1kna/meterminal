draw: function(ctx, offsetX, offsetY, mood) {
  if (!e.u46 || ctx !== facectx) return;  // только в спец-режиме, только на face canvas

  const fill = (x, y, w, h, color) => {
    ctx.fillStyle = color;
    ctx.fillRect(offsetX + Math.round(x), offsetY + Math.round(y), w, h);
  };

  const moodOffset = {
    surprised: -3, heart: -2, grin: -1,
    smile: 0, wink: 0,
    skeptic: 1, sad: 2, angry: 2, u47: -3
  }[mood] || 0;

  for (const [x, y, w, h, color] of (t.s06?.cap || [])) {
    fill(x, y + moodOffset, w, h, color);
  }

  if (mood !== "angry" && mood !== "u47") {
    fill(19, 10, 20, 2, i.light);   // левая бровь
    fill(61, 10, 20, 2, i.light);   // правая бровь
  }

  const canMove = !e.blink && mood !== "wink" && mood !== "u47";
  const eyeLeftX = 28 + (canMove ? e.eyeX : 0);
  const eyeTopY  = (mood === "u47" ? 28 : 24) + (canMove ? e.eyeY : 0);

  for (const [x, y, w, h] of (t.s06?.rim || [])) {
    fill(eyeLeftX + x, eyeTopY + y, w, h, "#d2a55c");
  }

  fill(eyeLeftX - 9, eyeTopY - 7, 2, 3, "#a7adb0");  // светлый блик
  fill(eyeLeftX - 10, eyeTopY - 3, 2, 2, "#737d80"); // тень

  //    в этом файле правый глаз НЕ рисуется — только левый
  //    (видимо, rim уже содержит формы для обоих глаз)

  ctx.fillStyle = i.light;  // восстановить цвет
}
